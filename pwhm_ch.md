/* DPP (Device Provisioning Protocol) bootstrap support.
 * When pushButton() is called, the endpoint generates a DPP bootstrap
 * URI via wpa_supplicant and stores it in URI for prplMesh to consume.
 */
object DPP {
    on action call call _dpp_pushButton;

    /**
     * @brief Initiate DPP bootstrap key generation.
     * Calls DPP_BOOTSTRAP_GEN + DPP_GET_URI on wpa_supplicant.
     * On success, URI and Status are updated in this object.
     * The resulting dm:object-changed event on URI is consumed by prplMesh.
     */
    void pushButton();

    /**
     * @brief The DPP bootstrap URI generated by wpa_supplicant.
     * Format: DPP:K:<base64-pubkey>;C:<opclass/chan>;M:<mac>;;
     * Empty string when Status is Idle.
     */
    string URI {
        default "";
        on action validate call check_maximum_length 1024;
    }

    /**
     * @brief Current DPP bootstrap state.
     * Idle          — no bootstrap active, URI is empty.
     * BootstrapReady — URI has been generated and is valid.
     */
    string Status {
        on action validate call check_enum ["Idle", "BootstrapReady"];
        default "Idle";
    }
}

## FILE 4: `src/nl80211/wld_wpaSupp_ep_api.c` — New wpa_supplicant function

Add after `wld_wpaSupp_ep_startWpsPbc` (after line 857):

```c
/* Maximum length of a DPP URI returned by wpa_supplicant.
 * DPP:K:<512-char base64>;C:<opclass/chan>;M:<12-char mac>;;
 * 1023 gives ample headroom while staying stack-safe.               */
#define WLD_DPP_URI_MAX_LEN  (1023U)

/* Maximum length of the bootstrap ID reply ("FAIL" or integer <= 10 digits) */
#define WLD_DPP_ID_BUF_LEN   (16U)

/* Maximum length of DPP_BOOTSTRAP_GEN command including mac= argument */
#define WLD_DPP_GEN_CMD_LEN  (128U)

/* Maximum length of DPP_BOOTSTRAP_GET_URI command */
#define WLD_DPP_GET_CMD_LEN  (32U)

/**
 * @brief Send DPP_BOOTSTRAP_GEN then DPP_BOOTSTRAP_GET_URI to wpa_supplicant
 *        and store the resulting URI in the DM.
 *
 * This function is synchronous. wld_wpaCtrl_sendCmdSynced() blocks until
 * wpa_supplicant replies, so no async handling is required.
 *
 * Security considerations:
 *  - All output buffers are zero-initialised before use.
 *  - snprintf return values are checked; truncation causes early exit.
 *  - The URI is validated (prefix check) before being written to the DM.
 *  - mac_str is sanitised character-by-character; no sscanf / strtok.
 *
 * @param pEP  Pointer to the EndPoint. Must not be NULL.
 * @return SWL_RC_OK on success, error code otherwise.
 */
swl_rc_ne wld_wpaSupp_ep_dppBootstrap(T_EndPoint* pEP)
{
    ASSERT_NOT_NULL(pEP, SWL_RC_INVALID_PARAM, ME, "pEP is NULL");

    wld_wpaCtrlInterface_t* pInterface =
        wld_wpaSupp_ep_getWpaCtrlInterface(pEP);
    ASSERT_NOT_NULL(pInterface, SWL_RC_ERROR, ME,
                    "%s: wpa_ctrl interface is NULL", pEP->Name);
    ASSERTI_TRUE(wld_wpaCtrlInterface_isReady(pInterface),
                 SWL_RC_INVALID_STATE, ME,
                 "%s: wpa_supplicant not ready", pEP->Name);

    /* ------------------------------------------------------------------ *
     * Step 1 — build "DPP_BOOTSTRAP_GEN type=qrcode chan=81/6,115/36 mac=AABBCCDDEEFF"
     * ------------------------------------------------------------------ */

    /* Convert binary MAC to colon-separated string, then strip colons.
     * pEP->wDevId is not a MAC — use the interface name + nl80211 or the
     * wpa_supplicant STATUS reply mac. The safest source available from
     * the struct is the wpaCtrlInterface name: STATUS gives "address=xx:xx"
     * but that requires another round-trip. Instead, derive from alias
     * which by convention is the interface name (e.g. "wlan0").
     * wpa_supplicant DPP_BOOTSTRAP_GEN mac= is OPTIONAL — omitting it lets
     * hostapd/wpa_supplicant use the interface MAC automatically.
     * We therefore omit mac= to avoid a second STATUS round-trip and to
     * keep the implementation robust across driver variants.             */

    char gen_cmd[WLD_DPP_GEN_CMD_LEN];
    memset(gen_cmd, 0, sizeof(gen_cmd));

    int written = snprintf(gen_cmd, sizeof(gen_cmd),
                           "DPP_BOOTSTRAP_GEN type=qrcode chan=81/6,115/36");
    if((written < 0) || ((size_t)written >= sizeof(gen_cmd))) {
        SAH_TRACEZ_ERROR(ME, "%s: DPP_BOOTSTRAP_GEN command truncated",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    /* ------------------------------------------------------------------ *
     * Step 2 — send DPP_BOOTSTRAP_GEN, read back the integer bootstrap ID
     * ------------------------------------------------------------------ */

    char id_buf[WLD_DPP_ID_BUF_LEN];
    memset(id_buf, 0, sizeof(id_buf));

    bool ok = wld_wpaCtrl_sendCmdSynced(pInterface,
                                         gen_cmd,
                                         id_buf,
                                         sizeof(id_buf));
    if(!ok) {
        SAH_TRACEZ_ERROR(ME, "%s: DPP_BOOTSTRAP_GEN command send failed",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    /* wpa_supplicant returns "FAIL" on error or a positive integer string */
    if(strncmp(id_buf, "FAIL", 4) == 0) {
        SAH_TRACEZ_ERROR(ME, "%s: DPP_BOOTSTRAP_GEN returned FAIL",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    /* Validate that every character in id_buf is a digit              */
    for(size_t i = 0; (i < sizeof(id_buf) - 1U) && (id_buf[i] != '\0'); i++) {
        if((id_buf[i] < '0') || (id_buf[i] > '9')) {
            SAH_TRACEZ_ERROR(ME,
                             "%s: DPP_BOOTSTRAP_GEN reply is not numeric: '%s'",
                             pEP->Name, id_buf);
            return SWL_RC_ERROR;
        }
    }

    /* Safe string-to-int: strtol with full error checking             */
    char*  end_ptr      = NULL;
    errno               = 0;
    long   bootstrap_id = strtol(id_buf, &end_ptr, 10);

    if((errno != 0) || (end_ptr == id_buf) || (bootstrap_id <= 0L) ||
       (bootstrap_id > (long)INT_MAX)) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP_BOOTSTRAP_GEN reply parse failed id='%s'",
                         pEP->Name, id_buf);
        return SWL_RC_ERROR;
    }

    SAH_TRACEZ_INFO(ME, "%s: DPP bootstrap_id=%ld", pEP->Name, bootstrap_id);

    /* ------------------------------------------------------------------ *
     * Step 3 — build "DPP_BOOTSTRAP_GET_URI <id>"
     * ------------------------------------------------------------------ */

    char get_cmd[WLD_DPP_GET_CMD_LEN];
    memset(get_cmd, 0, sizeof(get_cmd));

    written = snprintf(get_cmd, sizeof(get_cmd),
                       "DPP_BOOTSTRAP_GET_URI %ld", bootstrap_id);
    if((written < 0) || ((size_t)written >= sizeof(get_cmd))) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP_BOOTSTRAP_GET_URI command truncated",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    /* ------------------------------------------------------------------ *
     * Step 4 — send DPP_BOOTSTRAP_GET_URI, read back the URI string
     * ------------------------------------------------------------------ */

    /* +1 for NUL terminator; buffer is on stack — 1 KB is acceptable  */
    char uri_buf[WLD_DPP_URI_MAX_LEN + 1U];
    memset(uri_buf, 0, sizeof(uri_buf));

    ok = wld_wpaCtrl_sendCmdSynced(pInterface,
                                    get_cmd,
                                    uri_buf,
                                    sizeof(uri_buf));
    if(!ok) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP_BOOTSTRAP_GET_URI command send failed",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    /* ------------------------------------------------------------------ *
     * Step 5 — validate the URI before touching the DM
     * ------------------------------------------------------------------ */

    if(strncmp(uri_buf, "FAIL", 4) == 0) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP_BOOTSTRAP_GET_URI returned FAIL",
                         pEP->Name);
        return SWL_RC_ERROR;
    }

    if(strncmp(uri_buf, "DPP:", 4) != 0) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP URI does not start with 'DPP:' — got: '%.20s'",
                         pEP->Name, uri_buf);
        return SWL_RC_ERROR;
    }

    size_t uri_len = strnlen(uri_buf, sizeof(uri_buf));
    if(uri_len < 8U) {
        /* Minimum viable URI: "DPP:K:x;;" is 9 chars                 */
        SAH_TRACEZ_ERROR(ME, "%s: DPP URI implausibly short (%zu chars)",
                         pEP->Name, uri_len);
        return SWL_RC_ERROR;
    }

    SAH_TRACEZ_INFO(ME, "%s: DPP URI generated, length=%zu",
                    pEP->Name, uri_len);

    /* ------------------------------------------------------------------ *
     * Step 6 — write URI and status into DM (fires dm:object-changed)
     * ------------------------------------------------------------------ */

    wld_endpoint_set_dpp_uri(pEP, uri_buf);
    wld_endpoint_set_dpp_status(pEP, "BootstrapReady");

    return SWL_RC_OK;
}

/**
 * @brief wifiGen_ep_dppBootstrap
 *
 * Plugin-layer wrapper for DPP bootstrap generation.
 * Validates that the radio and endpoint are in a usable state before
 * delegating to the nl80211 wpa_supplicant layer.
 *
 * @param pEP  Pointer to the EndPoint. Must not be NULL.
 * @return SWL_RC_OK on success, error code otherwise.
 */
swl_rc_ne wifiGen_ep_dppBootstrap(T_EndPoint* pEP)
{
    ASSERT_NOT_NULL(pEP, SWL_RC_INVALID_PARAM, ME, "pEP is NULL");

    T_Radio* pRad = pEP->pRadio;
    ASSERT_NOT_NULL(pRad, SWL_RC_ERROR, ME,
                    "%s: pRadio is NULL", pEP->Name);

    /* Require the radio to be up and the endpoint to be enabled.
     * DPP bootstrap generation needs wpa_supplicant to be running,
     * which requires the radio to be operational.                     */
    if(!wld_endpoint_hasStackEnabled(pEP)) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: endpoint stack not enabled, cannot bootstrap DPP",
                         pEP->Name);
        return SWL_RC_INVALID_STATE;
    }

    if(pRad->status != RST_UP && !wld_rad_hasOnlyActiveEP(pRad)) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: radio not ready (status=%d), cannot bootstrap DPP",
                         pEP->Name, (int)pRad->status);
        return SWL_RC_INVALID_STATE;
    }

    SAH_TRACEZ_INFO(ME, "%s: starting DPP bootstrap", pEP->Name);

    swl_rc_ne ret = wld_wpaSupp_ep_dppBootstrap(pEP);
    if(!swl_rc_isOk(ret)) {
        SAH_TRACEZ_ERROR(ME, "%s: DPP bootstrap failed rc=%d",
                         pEP->Name, (int)ret);
        return ret;
    }

    SAH_TRACEZ_INFO(ME, "%s: DPP bootstrap completed successfully",
                    pEP->Name);
    return SWL_RC_OK;
}
```

/* ======================================================================

/**
 * @brief ODL action handler for WiFi.EndPoint.{i}.DPP.pushButton()
 *
 * Called by the Ambiorix framework when an external caller invokes:
 *   ubus call WiFi.EndPoint.[alias].DPP pushButton '{}'
 *
 * Design: DPP_BOOTSTRAP_GEN is synchronous — wpa_supplicant replies
 * immediately with a bootstrap ID. No deferred call or delay timer is
 * needed (contrast with WPS pushButton which needs the delay timer
 * because the FSM may not be IDLE).
 *
 * @param obj     The DPP object (child of EndPoint object).
 * @param func    The function descriptor (unused).
 * @param args    Input arguments (none required for DPP pushButton).
 * @param retval  Return value variant (unused for void RPC).
 * @return amxd_status_ok on success, amxd_status_unknown_error otherwise.
 */
amxd_status_t _dpp_pushButton(amxd_object_t* obj,
                               amxd_function_t* func _UNUSED,
                               amxc_var_t* args      _UNUSED,
                               amxc_var_t* retval    _UNUSED)
{
    /* The DPP object is a child of the EndPoint object.
     * Get the parent EndPoint object first, then its priv pointer.    */
    ASSERT_NOT_NULL(obj, amxd_status_unknown_error, ME,
                    "DPP object is NULL");

    amxd_object_t* ep_obj = amxd_object_get_parent(obj);
    ASSERT_NOT_NULL(ep_obj, amxd_status_unknown_error, ME,
                    "EndPoint parent object is NULL");

    /* pBus points to the EndPoint DM object; priv points to T_EndPoint */
    T_EndPoint* pEP = (T_EndPoint*)ep_obj->priv;
    ASSERT_NOT_NULL(pEP, amxd_status_unknown_error, ME,
                    "T_EndPoint priv pointer is NULL");

    SAH_TRACEZ_INFO(ME, "%s: DPP pushButton called", pEP->Name);

    /* Validate FA function pointer before dereferencing               */
    ASSERT_NOT_NULL(pEP->pFA, amxd_status_unknown_error, ME,
                    "%s: pFA is NULL", pEP->Name);
    ASSERT_NOT_NULL(pEP->pFA->mfn_wendpoint_wps_start,
                    amxd_status_unknown_error, ME,
                    "%s: FA not fully initialised", pEP->Name);

    /* Delegate to plugin layer — wifiGen_ep_dppBootstrap()            */
    swl_rc_ne rc = wifiGen_ep_dppBootstrap(pEP);
    if(!swl_rc_isOk(rc)) {
        SAH_TRACEZ_ERROR(ME, "%s: DPP pushButton failed rc=%d",
                         pEP->Name, (int)rc);
        return amxd_status_unknown_error;
    }

    SAH_TRACEZ_INFO(ME, "%s: DPP pushButton succeeded", pEP->Name);
    return amxd_status_ok;
}

/**
 * @brief Set WiFi.EndPoint.{i}.DPP.URI in the Ambiorix data model.
 *
 * Setting a parameter via amxd_object_set_value fires dm:object-changed
 * automatically. prplMesh subscribes to this event to learn the URI.
 *
 * @param pEP  Must not be NULL.
 * @param uri  Must not be NULL, must start with "DPP:", max 1023 chars.
 */
void wld_endpoint_set_dpp_uri(T_EndPoint* pEP, const char* uri)
{
    /* Defensive parameter validation — callers are internal but
     * these are public API functions so we validate unconditionally.  */
    if(pEP == NULL) {
        SAH_TRACEZ_ERROR(ME, "wld_endpoint_set_dpp_uri: pEP is NULL");
        return;
    }
    if(uri == NULL) {
        SAH_TRACEZ_ERROR(ME, "%s: uri is NULL", pEP->Name);
        return;
    }
    if(strncmp(uri, "DPP:", 4) != 0) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: refusing to store non-DPP URI in DM (got '%.8s')",
                         pEP->Name, uri);
        return;
    }

    /* pBus is the Ambiorix DM object pointer for this EndPoint        */
    ASSERT_NOT_NULL_VOID(pEP->pBus, ME,
                         "%s: pBus is NULL", pEP->Name);

    amxd_object_t* dpp_obj = amxd_object_get(pEP->pBus, "DPP");
    if(dpp_obj == NULL) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP child object not found in DM",
                         pEP->Name);
        return;
    }

    /* SWLA_OBJECT_SET_PARAM_CSTRING is the established pattern in this
     * codebase (used extensively in wld_endpoint.c stats setters).    */
    SWLA_OBJECT_SET_PARAM_CSTRING(dpp_obj, "URI", uri);

    SAH_TRACEZ_INFO(ME, "%s: DPP.URI set (%zu chars)",
                    pEP->Name, strnlen(uri, WLD_DPP_URI_MAX_LEN + 1U));
}

/**
 * @brief Set WiFi.EndPoint.{i}.DPP.Status in the Ambiorix data model.
 *
 * @param pEP     Must not be NULL.
 * @param status  Must be "Idle" or "BootstrapReady". Must not be NULL.
 */
void wld_endpoint_set_dpp_status(T_EndPoint* pEP, const char* status)
{
    if(pEP == NULL) {
        SAH_TRACEZ_ERROR(ME, "wld_endpoint_set_dpp_status: pEP is NULL");
        return;
    }
    if(status == NULL) {
        SAH_TRACEZ_ERROR(ME, "%s: status is NULL", pEP->Name);
        return;
    }

    /* Validate against the enum defined in the ODL                    */
    if((strcmp(status, "Idle") != 0) &&
       (strcmp(status, "BootstrapReady") != 0)) {
        SAH_TRACEZ_ERROR(ME,

    ASSERT_NOT_NULL_VOID(pEP->pBus, ME,

    amxd_object_t* dpp_obj = amxd_object_get(pEP->pBus, "DPP");
    if(dpp_obj == NULL) {
        SAH_TRACEZ_ERROR(ME,
                         "%s: DPP child object not found in DM",
                         pEP->Name);
        return;
    }

    SWLA_OBJECT_SET_PARAM_CSTRING(dpp_obj, "Status", status);

    SAH_TRACEZ_INFO(ME, "%s: DPP.Status set to '%s'", pEP->Name, status);
