bool controller_ucc_listener::handle_dev_exec_action(
    const std::unordered_map<std::string, std::string> &params, std::string &err_string)
{
    auto action_type_it = params.find("dppactiontype");
    if (action_type_it != params.end()) {
        auto &action_type = action_type_it->second;
        if (action_type != "SetPeerBootstrap") {
            err_string = "DPPActionType is not 'SetPeerBootstrap', value is '" + action_type + "'";
            return false;
        }

        // Extract alias — mandatory for map keying
        auto alias_it = params.find("alias");
        if (alias_it == params.end() || alias_it->second.empty()) {
            err_string = "DPP alias is missing or empty";
            return false;
        }
        const std::string &alias = alias_it->second;

        // Extract bootstrapping data
        auto dpp_bootstrapping_data_bytes_it = params.find("dppbootstrappingdata");
        if (dpp_bootstrapping_data_bytes_it == params.end()) {
            err_string = "DPPBootstrappingData is missing";
            return false;
        }

        // Decode hex bytes to DPP URI string
        auto dpp_uri =
            beerocks::string_utils::bytes_string_to_string(
                dpp_bootstrapping_data_bytes_it->second);

        LOG(DEBUG) << "alias: " << alias;
        LOG(DEBUG) << "bootstrapping data decoded to DPP URI: " << dpp_uri;

        // Delegate entirely to dpp_uri_parse — no duplicate logic
        if (!m_database.dpp_uri_parse(alias, dpp_uri)) {
            err_string = "Failed to parse DPP URI for alias: " + alias;
            return false;
        }

        return true;
    }

    err_string = "command is not supported on the Controller";
    return false;
}

The Problem We Found
The original code had a single struct instance in db.h:
This means only one enrollee could be stored at a time — any new bootstrapping info would silently overwrite the previous one. In a real EasyMesh network with multiple enrollees, this is a bug.
What We Changed and Why
1. Single struct → Map
We replaced the single instance with a map keyed by alias, so multiple enrollees can be stored simultaneously.
2. Single parser for both paths
Both paths deliver the same data — a DPP URI string. So we have one shared parser dpp_uri_parse() that both paths call. No duplicated logic.
3. Alias handling
ACS path — alias comes from ACS, TR-181 compliant, trusted as-is
UCC path — we simply require alias as an additional parameter in the UCC payload. The test harness can trivially provide it since it's a flexible key-value command
