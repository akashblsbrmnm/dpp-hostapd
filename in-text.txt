// NOTE: calculate_dpp_bootstrapping_str() still works — it can be updated
// later to take an alias parameter if needed. For now it reads from the
// map by alias if required, or can be left as-is for backward compat.

bool db::dpp_uri_parse(const std::string &alias, const std::string &dpp_uri)
{
    const std::string prefix = "DPP:";
    if (dpp_uri.compare(0, prefix.size(), prefix) != 0) {
        LOG(ERROR) << "Invalid DPP URI prefix: " << dpp_uri;
        return false;
    }

    sDppBootstrappingInfo dpp_info;
    dpp_info.alias = alias;

    std::string body = dpp_uri.substr(prefix.size());
    std::stringstream ss(body);
    std::string token;

    while (std::getline(ss, token, ';')) {
        if (token.empty()) {
            continue;
        }

        auto sep = token.find(':');
        if (sep == std::string::npos) {
            continue;
        }

        std::string key = token.substr(0, sep);
        std::string val = token.substr(sep + 1);

        if (key == "C") {
            std::stringstream cs(val);
            std::string entry;
            while (std::getline(cs, entry, ',')) {
                auto slash = entry.find('/');
                if (slash == std::string::npos) {
                    continue;
                }
                try {
                    uint8_t op_class = static_cast<uint8_t>(std::stoi(entry.substr(0, slash)));
                    uint8_t channel  = static_cast<uint8_t>(std::stoi(entry.substr(slash + 1)));
                    dpp_info.operating_class_channel.emplace(op_class, channel);
                } catch (const std::exception &e) {
                    LOG(ERROR) << "Invalid C field in DPP URI: " << entry;
                }
            }
        } else if (key == "M") {
            std::string mac_with_colons;
            for (size_t i = 0; i < val.size(); i += 2) {
                if (i > 0) {
                    mac_with_colons += ':';
                }
                mac_with_colons += val.substr(i, 2);
            }
            dpp_info.mac = tlvf::mac_from_string(mac_with_colons);
        } else if (key == "I") {
            dpp_info.info = val;
        } else if (key == "V") {
            try {
                dpp_info.version = static_cast<uint8_t>(std::stoi(val));
            } catch (const std::exception &e) {
                LOG(ERROR) << "Invalid V field in DPP URI: " << val;
            }
        } else if (key == "H") {
            dpp_info.host = val;
        } else if (key == "K") {
            dpp_info.public_key = val;
        }
    }

    if (dpp_info.public_key.empty()) {
        LOG(ERROR) << "DPP URI missing mandatory public key (K field) for alias: " << alias;
        return false;
    }

    // Insert or overwrite entry for this Alias
    dpp_bootstrapping_map[alias] = std::move(dpp_info);
    LOG(DEBUG) << "Stored DPP bootstrapping info for alias: " << alias
               << ", total entries: " << dpp_bootstrapping_map.size();

    return true;
}

void db::remove_dpp_bootstrapping_info(const std::string &alias)
{
    auto it = dpp_bootstrapping_map.find(alias);
    if (it != dpp_bootstrapping_map.end()) {
        dpp_bootstrapping_map.erase(it);
        LOG(DEBUG) << "Removed DPP bootstrapping info for alias: " << alias;
    }
}

const std::unordered_map<std::string, db::sDppBootstrappingInfo> &
db::get_dpp_bootstrapping_map() const
{
    return dpp_bootstrapping_map;
}


// -----------------------------------------------------------------------
// on_action.cpp — updated event_provisioned_dpp_changed
// Now reads Alias too, passes both to dpp_uri_parse
// Also handles instance removal via "dm:instance-removed" signal
// -----------------------------------------------------------------------

static void event_provisioned_dpp_changed(const char *const sig_name, const amxc_var_t *const data,
                                          void *const priv)
{
    amxd_object_t *dpp_obj =
        amxd_dm_signal_get_object(beerocks::nbapi::Amxrt::getDatamodel(), data);

    if (!dpp_obj) {
        LOG(WARNING) << "Failed to get ProvisionedDPP object";
        return;
    }

    std::string alias   = get_param_string(dpp_obj, "Alias");
    std::string dpp_uri = get_param_string(dpp_obj, "DPP_URI");

    if (alias.empty()) {
        LOG(WARNING) << "ProvisionedDPP Alias is empty, skipping";
        return;
    }

    if (dpp_uri.empty()) {
        LOG(WARNING) << "DPP_URI is empty for alias: " << alias;
        return;
    }

    if (!g_database->dpp_uri_parse(alias, dpp_uri)) {
        LOG(ERROR) << "Failed to parse DPP_URI for alias: " << alias;
    }
}

static void event_provisioned_dpp_removed(const char *const sig_name, const amxc_var_t *const data,
                                          void *const priv)
{
    // When an ACS deletes a ProvisionedDPP instance, remove it from db too
    amxd_object_t *dpp_obj =
        amxd_dm_signal_get_object(beerocks::nbapi::Amxrt::getDatamodel(), data);

    if (!dpp_obj) {
        LOG(WARNING) << "Failed to get ProvisionedDPP object on removal";
        return;
    }

    std::string alias = get_param_string(dpp_obj, "Alias");
    if (alias.empty()) {
        LOG(WARNING) << "ProvisionedDPP Alias is empty on removal, skipping";
        return;
    }

    g_database->remove_dpp_bootstrapping_info(alias);
}

// -----------------------------------------------------------------------
// get_events_list() — register both handlers
// -----------------------------------------------------------------------

std::vector<beerocks::nbapi::sEvents> get_events_list(void)
{
    const std::vector<beerocks::nbapi::sEvents> events_list = {
        {"event_configuration_changed", event_configuration_changed},
        {"event_traffic_separation_changed", event_traffic_separation_changed},
        {"event_network_group_changed", event_network_group_changed},
        {"event_network_enable_changed", event_network_enable_changed},
        {"event_provisioned_dpp_changed", event_provisioned_dpp_changed},
        {"event_provisioned_dpp_removed", event_provisioned_dpp_removed}};
    return events_list;
}

   controller.odl.in — ProvisionedDPP object with events wired
   ----------------------------------------------------------------------- */

                    /* DPP Provisioning */
                    object ProvisionedDPP[] {
                        counted with ProvisionedDPPNumberOfEntries;
                        %persistent string Alias = "";
                        %persistent string DPP_URI = "";
                        on event "dm:object-changed" call event_provisioned_dpp_changed;
                        on event "dm:instance-removed" call event_provisioned_dpp_removed;
                    }

/* -----------------------------------------------------------------------
   Why two events:

   "dm:object-changed"  — fires when ACS writes DPP_URI to an existing
                          instance → parse and store in db

   "dm:instance-removed" — fires when ACS deletes a ProvisionedDPP instance
                           → remove from db so stale URIs don't persist
   ----------------------------------------------------------------------- */
