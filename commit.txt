static swl_rc_ne s_sendHostapdReply(T_AccessPoint* pAP,
                                    char* reply,
                                    size_t replySize,
                                    const char* reason,
                                    const char* format,
                                    ...) {
    ASSERTS_NOT_NULL(pAP, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_NOT_NULL(reply, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_NOT_NULL(format, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_TRUE(replySize > 0, SWL_RC_INVALID_PARAM, ME, "invalid reply size");

    ASSERTS_TRUE(wld_wpaCtrlInterface_isReady(pAP->wpaCtrlInterface),
                 SWL_RC_INVALID_STATE,
                 ME,
                 "%s: wpactrl link not ready",
                 pAP->alias);

    char cmd[256] = {0};

    va_list args;
    va_start(args, format);
    int len = vsnprintf(cmd, sizeof(cmd), format, args);
    va_end(args);

    ASSERT_TRUE((len >= 0) && ((size_t) len < sizeof(cmd)),
                SWL_RC_ERROR,
                ME,
                "%s: command truncated",
                pAP->alias);

    SAH_TRACEZ_INFO(ME,
                    "%s: send hostapd cmd %s for %s",
                    wld_wpaCtrlInterface_getName(pAP->wpaCtrlInterface),
                    cmd,
                    reason ? reason : "unknown");

    bool ret = wld_wpaCtrl_sendCmdSynced(pAP->wpaCtrlInterface,
                                         cmd,
                                         reply,
                                         replySize);

    ASSERT_TRUE(ret,
                SWL_RC_ERROR,
                ME,
                "%s: hostapd command failed (%s)",
                pAP->alias,
                cmd);

    swl_str_removeWhitespaceEdges(reply);

    SAH_TRACEZ_INFO(ME,
                    "%s: hostapd reply [%s]",
                    pAP->alias,
                    reply);

    return SWL_RC_OK;
}
swl_rc_ne wld_ap_hostapd_dppBootstrapGen(T_AccessPoint* pAP,
                                         const char* params,
                                         uint32_t* bootstrapId);
swl_rc_ne wld_ap_hostapd_dppBootstrapGen(T_AccessPoint* pAP,
                                         const char* params,
                                         uint32_t* bootstrapId) {
    ASSERTS_NOT_NULL(pAP, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_NOT_NULL(bootstrapId, SWL_RC_INVALID_PARAM, ME, "NULL");

    char reply[32] = {0};

    swl_rc_ne rc =
        s_sendHostapdReply(
            pAP,
            reply,
            sizeof(reply),
            "DPP_BOOTSTRAP_GEN",
            swl_str_isEmpty(params) ?
                "DPP_BOOTSTRAP_GEN" :
                "DPP_BOOTSTRAP_GEN %s",
            params);

    ASSERT_TRUE(swl_rc_isOk(rc),
                rc,
                ME,
                "%s: DPP_BOOTSTRAP_GEN failed",
                pAP->alias);

    /*
     * Expected replies:
     *   "1"
     *   "15"
     *   "123"
     * Failure:
     *   "FAIL"
     */

    errno = 0;

    char* end = NULL;
    long id = strtol(reply, &end, 10);

    ASSERT_TRUE(errno == 0 &&
                end != reply &&
                *end == '\0' &&
                id > 0 &&
                id <= UINT32_MAX,
                SWL_RC_ERROR,
                ME,
                "%s: invalid bootstrap id reply [%s]",
                pAP->alias,
                reply);

    *bootstrapId = (uint32_t) id;

    SAH_TRACEZ_INFO(ME,
                    "%s: DPP bootstrap id %u",
                    pAP->alias,
                    *bootstrapId);

    return SWL_RC_OK;
}
