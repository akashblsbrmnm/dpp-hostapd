// Guard to prevent boot-looping the event handler
bool g_templates_dm_initialized = false;

// Helper to handle whitespace in INI sections
static inline void trim_whitespace(std::string &s) {
    s.erase(s.find_last_not_of(" \t\r\n") + 1);
    s.erase(0, s.find_first_not_of(" \t\r\n"));
}

static std::string get_effective_controller_config() {
    std::string fallback = mapf::utils::get_install_path() + "config/beerocks_controller.conf";

#ifdef BEEROCKS_RDKB
    std::string primary = "/nvram/beerocks_controller.conf";
#else
    std::string primary = "./beerocks_controller.conf";
#endif

    constexpr char TMP_CONF[] = "/tmp/beerocks_controller.conf";
    if (access(TMP_CONF, F_OK) == 0) {
        return TMP_CONF;
    }
    if (access(primary.c_str(), F_OK) == 0) {
        return primary;
    }
    return fallback;
}

static bool update_master_config_file(bool enable_templates) {
    std::string target_file = get_effective_controller_config();
    std::ifstream file_in(target_file);
    
    if (!file_in.is_open()) {
        LOG(ERROR) << "Failed to open config file: " << target_file;
        return false;
    }

    std::vector<std::string> lines;
    std::string line;
    bool key_found = false;
    bool in_controller_sec = false;
    size_t insert_pos = std::string::npos;
    
    std::string target_key = "use_dataelements_vap_configs=";
    std::string new_val = enable_templates ? "1" : "0";

    while (std::getline(file_in, line)) {
        std::string trimmed = line;
        trim_whitespace(trimmed);

        if (trimmed == "[controller]") {
            in_controller_sec = true;
        } else if (!trimmed.empty() && trimmed[0] == '[') {
            if (in_controller_sec) {
                insert_pos = lines.size();
                in_controller_sec = false;
            }
        }

        if (in_controller_sec && trimmed.compare(0, target_key.size(), target_key) == 0) {
            line = target_key + new_val;
            key_found = true;
        }
        lines.push_back(line);
    }
    file_in.close();

    if (in_controller_sec && insert_pos == std::string::npos) {
        insert_pos = lines.size();
    }

    if (!key_found) {
        if (insert_pos != std::string::npos) {
            lines.insert(lines.begin() + insert_pos, target_key + new_val);
        } else {
            lines.push_back("[controller]");
            lines.push_back(target_key + new_val);
        }
    }

    std::string tmp_file = target_file + ".tmp";
    std::ofstream file_out(tmp_file, std::ios::trunc);
    if (!file_out.is_open()) {
        LOG(ERROR) << "Failed to open tmp file: " << tmp_file;
        return false;
    }

    for (const auto& l : lines) {
        file_out << l << "\n";
    }
    
    // Flush to OS buffers before the atomic rename
    file_out.flush();
    file_out.close();

    if (std::rename(tmp_file.c_str(), target_file.c_str()) != 0) {
        LOG(ERROR) << "Failed to rename tmp config file";
        std::remove(tmp_file.c_str());
        return false;
    }

    LOG(INFO) << "Persisted " << target_key << new_val << " into " << target_file;
    return true;
}

static void event_use_dataelements_vap_config_changed(
    const char *const sig_name,
    const amxc_var_t *const data,
    void *const priv)
{
    // Ignore events fired during startup synchronization
    if (!g_templates_dm_initialized) {
        return;
    }

    auto *network = amxd_dm_signal_get_object(beerocks::nbapi::Amxrt::getDatamodel(), data);
    if (!network) {  
        LOG(ERROR) << "Failed to get DataElements.Network";  
        return;  
    }  

    bool enabled = get_param_bool(network, "UseDataElementsVapConfigs");  

    /*
     * Persist DM value into beerocks_controller.conf.
     * Runtime behavior remains unchanged.
     * Controller restart is required for the new configuration to take effect.
     */
    if (!update_master_config_file(enabled)) {
        LOG(ERROR) << "Failed to apply template configuration change to controller config.";
        return;
    }

    LOG(INFO) << "Configuration updated. Restart the prplMesh controller to apply changes.";
}
