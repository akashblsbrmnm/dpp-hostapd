bool Controller::handle_cmdu_1905_dpp_bootstrapping_uri_notification(
    const sMacAddr &src_mac, ieee1905_1::CmduMessageRx &cmdu_rx)
{
    auto uri_tlv = cmdu_rx.getClass<wfa_map::tlvDppBootstrappingUriNotification>();
    if (!uri_tlv) {
        LOG(ERROR) << "wfa_map::tlvDppBootstrappingUriNotification missing";
        return false;
    }

    const auto dpp_uri = uri_tlv->dpp_uri_str();
    if (dpp_uri.empty()) {
        LOG(WARNING) << "DPP bootstrapping URI notification with empty URI";
        return false;
    }

    // 1. Auto-generate an alias using the TLV's Enrollee MAC
    sMacAddr enrollee_mac = uri_tlv->backhaul_sta_address();
    if (enrollee_mac == beerocks::net::network_utils::ZERO_MAC) {
        enrollee_mac = uri_tlv->bssid(); // Fallback if bSTA MAC is missing
    }

    std::string mac_str = tlvf::mac_to_string(enrollee_mac);
    std::string alias = "auto-" + mac_str;
    
    // Optional: Clean colons from the alias (e.g. "auto-aa:bb" -> "auto-aabb")
    alias.erase(std::remove(alias.begin(), alias.end(), ':'), alias.end());

    // 2. Parse and store into the new map architecture
    std::string error;
    if (!m_database.parse_dpp_bootstrap_info(alias, dpp_uri, error)) {
        LOG(WARNING) << "Failed parsing DPP bootstrapping URI: " << error;
        return false;
    }

    // 3. Retrieve a pointer to the stored struct to update missing data
    auto info = m_database.get_dpp_bootstrap_info(alias);
    if (info) {
        // If the URI payload didn't contain an 'M:' field, populate it from the TLV
        if (info->mac == beerocks::net::network_utils::ZERO_MAC &&
            enrollee_mac != beerocks::net::network_utils::ZERO_MAC) {
            
            info->mac = enrollee_mac;
            LOG(DEBUG) << "Updated missing MAC in DPP info using TLV data";
        }

        LOG(INFO) << "Received DPP bootstrapping URI from proxy " << src_mac
                  << " ruid=" << tlvf::mac_to_string(uri_tlv->ruid())
                  << " bssid=" << tlvf::mac_to_string(uri_tlv->bssid())
                  << " backhaul_sta=" << tlvf::mac_to_string(uri_tlv->backhaul_sta_address())
                  << " generated_alias=" << alias
                  // We removed the hex string from the struct earlier to save memory, 
                  // so generate it here just for the log:
                  << " pkhash=" << hex_string_from_bytes(info->pkhash.data(), 32);
    }

    return true;
}
db::sDppBootstrappingInfo* db::get_dpp_bootstrap_info(const std::string &alias)
{
    auto it = dpp_bootstrap_info_map.find(alias);
    if (it != dpp_bootstrap_info_map.end()) {
        return &it->second;
    }
    return nullptr;
}
