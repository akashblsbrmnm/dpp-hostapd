static swl_rc_ne s_sendWpaSuppCommandReply(T_EndPoint* pEP,
                                           const char* cmd,
                                           char* reply,
                                           size_t replySize)
{
    ASSERTS_NOT_NULL(pEP, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_NOT_NULL(cmd, SWL_RC_INVALID_PARAM, ME, "NULL");
    ASSERTS_NOT_NULL(reply, SWL_RC_INVALID_PARAM, ME, "NULL");

    ASSERTS_TRUE(replySize > 0,
                 SWL_RC_INVALID_PARAM,
                 ME,
                 "invalid reply size");

    wld_wpaCtrlInterface_t* pInterface =
        wld_wpaSupp_ep_getWpaCtrlInterface(pEP);

    ASSERTS_TRUE(wld_wpaCtrlInterface_isReady(pInterface),
                 SWL_RC_INVALID_STATE,
                 ME,
                 "%s: Connection with wpa_supplicant not yet established",
                 pEP->Name);

    SAH_TRACEZ_INFO(ME,
                    "%s: send wpa_supplicant cmd [%s]",
                    pEP->Name,
                    cmd);

    bool ret =
        wld_wpaCtrl_sendCmdSynced(
            pInterface,
            cmd,
            reply,
            replySize);

    ASSERT_TRUE(ret,
                SWL_RC_ERROR,
                ME,
                "%s: command failed [%s]",
                pEP->Name,
                cmd);

    swl_str_removeWhitespaceEdges(reply);

    SAH_TRACEZ_INFO(ME,
                    "%s: reply [%s]",
                    pEP->Name,
                    reply);

    return SWL_RC_OK;
}

swl_rc_ne wld_wpaSupp_ep_dppBootstrapGen(
        T_EndPoint* pEP,
        const char* params,
        uint32_t* bootstrapId)
{
    ASSERTS_NOT_NULL(pEP,
                     SWL_RC_INVALID_PARAM,
                     ME,
                     "NULL");

    ASSERTS_NOT_NULL(bootstrapId,
                     SWL_RC_INVALID_PARAM,
                     ME,
                     "NULL");

    ASSERTS_FALSE(swl_str_isEmpty(params),
                  SWL_RC_INVALID_PARAM,
                  ME,
                  "%s: empty DPP parameters",
                  pEP->Name);

    ASSERTS_NOT_NULL(strstr(params, "type="),
                     SWL_RC_INVALID_PARAM,
                     ME,
                     "%s: mandatory parameter type= missing",
                     pEP->Name);

    char cmd[512] = {0};

    ASSERT_TRUE(
        swl_str_catFormat(
            cmd,
            sizeof(cmd),
            "DPP_BOOTSTRAP_GEN %s",
            params),
        SWL_RC_ERROR,
        ME,
        "command build failed");

    char reply[32] = {0};

    swl_rc_ne rc =
        s_sendWpaSuppCommandReply(
            pEP,
            cmd,
            reply,
            sizeof(reply));

    ASSERT_TRUE(
        swl_rc_isOk(rc),
        rc,
        ME,
        "%s: DPP_BOOTSTRAP_GEN failed",
        pEP->Name);

    ASSERT_FALSE(
        strcmp(reply, "FAIL") == 0,
        SWL_RC_ERROR,
        ME,
        "%s: DPP_BOOTSTRAP_GEN returned FAIL",
        pEP->Name);

    errno = 0;

    char* end = NULL;

    long id = strtol(reply, &end, 10);

    ASSERT_TRUE(
        errno == 0 &&
        end != reply &&
        *end == '\0' &&
        id > 0 &&
        id <= UINT32_MAX,
        SWL_RC_ERROR,
        ME,
        "%s: invalid bootstrap id reply [%s]",
        pEP->Name,
        reply);

    *bootstrapId = (uint32_t)id;

    return SWL_RC_OK;
}
swl_rc_ne wld_wpaSupp_ep_dppBootstrapGetUri(
        T_EndPoint* pEP,
        uint32_t bootstrapId,
        char* uri,
        size_t uriSize)
{
    ASSERTS_NOT_NULL(pEP,
                     SWL_RC_INVALID_PARAM,
                     ME,
                     "NULL");

    ASSERTS_NOT_NULL(uri,
                     SWL_RC_INVALID_PARAM,
                     ME,
                     "NULL");

    ASSERTS_TRUE(uriSize > 0,
                 SWL_RC_INVALID_PARAM,
                 ME,
                 "invalid uri size");

    ASSERTS_TRUE(bootstrapId > 0,
                 SWL_RC_INVALID_PARAM,
                 ME,
                 "invalid bootstrap id");

    char cmd[128] = {0};

    ASSERT_TRUE(
        swl_str_catFormat(
            cmd,
            sizeof(cmd),
            "DPP_BOOTSTRAP_GET_URI %u",
            bootstrapId),
        SWL_RC_ERROR,
        ME,
        "command build failed");

    swl_rc_ne rc =
        s_sendWpaSuppCommandReply(
            pEP,
            cmd,
            uri,
            uriSize);

    ASSERT_TRUE(
        swl_rc_isOk(rc),
        rc,
        ME,
        "%s: DPP_BOOTSTRAP_GET_URI failed",
        pEP->Name);

    ASSERT_FALSE(
        strcmp(uri, "FAIL") == 0,
        SWL_RC_ERROR,
        ME,
        "%s: DPP_BOOTSTRAP_GET_URI returned FAIL",
        pEP->Name);

    ASSERT_TRUE(
        strncmp(uri, "DPP:", 4) == 0,
        SWL_RC_ERROR,
        ME,
        "%s: invalid URI [%s]",
        pEP->Name,
        uri);

    return SWL_RC_OK;
}
amxd_status_t _EndPoint_dppBootstrapGen(
        amxd_object_t* object,
        amxd_function_t* func _UNUSED,
        amxc_var_t* args,
        amxc_var_t* retval)
{
    T_EndPoint* pEP = wld_ep_fromObj(
        amxd_object_get_parent(object));

    uint32_t bootstrapId = 0;

    const char* params =
        GET_CHAR_ARG(args, "params");

    swl_rc_ne rc =
        wld_wpaSupp_ep_dppBootstrapGen(
            pEP,
            params,
            &bootstrapId);

    ASSERT_TRUE(rc >= SWL_RC_OK,
                amxd_status_unknown_error,
                ME,
                "DPP bootstrap gen failed");

    amxc_var_add_key(
        uint32_t,
        retval,
        "BootstrapId",
        bootstrapId);

    return amxd_status_ok;
}
