%config {
    // Suppress duplication errors when merging into existing trees
    define-behavior = {
        existing-object = "update"
    };
}

%define {
    // Safely hook into the macro-generated path
    select "Device.WiFi" {

        %persistent object Templates {

            %persistent object Network {
                %persistent bool Enable = false;
                %persistent csv_string TopologyFlag = "";
                %persistent csv_string IEEE1905ALID = "";
                %persistent string PrimarySSCTemplateReference = "";
                %read-only string PrimarySSCTemplateID = "";
            }

            %persistent object BSSTemplate[] {
                counted with BSSTemplateNumberOfEntries;

                %key %persistent string Alias = "";
                %key %persistent string BSSTemplateID = "";
                %persistent uint32 Priority = 0 {
                    on action validate call check_range [0, 255];
                }
                %persistent string RadioTemplateReference = "";
                %persistent string SSCTemplateReference = "";
                %persistent string APMLDTemplateReference = "";
                %persistent string SecurityGroupReference = "";
                %read-only string LinkedRadioTemplateID = "";
                %read-only string LinkedSSCTemplateID = "";
                %read-only string LinkedAPMLDTemplateID = "";
                %read-only string LinkedSecurityGroupID = "";
                %persistent csv_string TopologyFlag = "";
                %persistent csv_string IEEE1905ALID = "";
                %persistent uint32 MaxWLANHopLimitFlag = 0 {
                    on action validate call check_range [0, 255];
                }
                %persistent string SecureEquipmentFlag = "null" {
                    default "null";
                    on action validate call check_enum [
                        "true", "false", "null"
                    ];
                }
                %persistent bool Enable = true;
                %persistent string SSID = "";
                %persistent string KeyPassphrase = "";
                %persistent bool AdvertisementEnable = true;
                %persistent string X_PRPLWARE-COM_VapType {
                    default "other";
                    on action validate call check_enum [
                        "home", "guest", "video", "backhaul",
                        "hotspot", "staff", "isolated", "other"
                    ];
                }
            }

            // ... (Include your other %persistent objects like RadioTemplate[], 
            // APMLDTemplate[], SSCTemplate[], SecurityGroup[], SecurityTemplate[] 
            // making sure every object declaration has the %persistent flag) ...

        } /* end Templates */
    } /* end select Device.WiFi */
}
