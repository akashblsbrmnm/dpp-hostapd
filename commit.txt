amxd_status_t provisioned_dpp_commit(amxd_object_t *object, amxd_function_t *func,
                                     amxc_var_t *args, amxc_var_t *ret)
{
    LOG(DEBUG) << "provisioned_dpp_commit called";

    amxd_object_t *dpp_table = amxd_object_get_child(object, "ProvisionedDPP");
    if (!dpp_table) {
        LOG(ERROR) << "ProvisionedDPP object not found";
        return amxd_status_object_not_found;
    }

    g_database->clear_dpp_bootstrapping_map();
    LOG(DEBUG) << "Cleared existing DPP bootstrapping map";

    amxd_object_for_each(instance, it, dpp_table)
    {
        amxd_object_t *inst = amxc_llist_it_get_data(it, amxd_object_t, it);

        std::string alias   = get_param_string(inst, "Alias");
        std::string dpp_uri = get_param_string(inst, "DPP_URI");

        LOG(DEBUG) << "Processing ProvisionedDPP instance, Alias: '" << alias
                   << "', DPP_URI: '" << dpp_uri << "'";

        if (alias.empty()) {
            LOG(WARNING) << "Skipping instance with empty Alias";
            continue;
        }

        if (dpp_uri.empty()) {
            LOG(WARNING) << "Skipping instance with empty DPP_URI for alias: " << alias;
            continue;
        }

        if (!g_database->dpp_uri_parse(alias, dpp_uri)) {
            LOG(ERROR) << "Failed to parse DPP_URI for alias: " << alias;
            continue;
        }
    }

    LOG(DEBUG) << "provisioned_dpp_commit done, total entries stored: "
               << g_database->get_dpp_bootstrapping_map().size();

    return amxd_status_ok;
}

// keep this for cleanup when instance is deleted
static void event_provisioned_dpp_removed(const char *const sig_name, const amxc_var_t *const data,
                                          void *const priv)
{
    amxd_object_t *dpp_obj =
        amxd_dm_signal_get_object(beerocks::nbapi::Amxrt::getDatamodel(), data);

    if (!dpp_obj) {
        LOG(WARNING) << "Failed to get ProvisionedDPP object on removal";
        return;
    }

    std::string alias = get_param_string(dpp_obj, "Alias");
    if (alias.empty()) {
        LOG(WARNING) << "ProvisionedDPP Alias is empty on removal, skipping";
        return;
    }

    LOG(DEBUG) << "Removing DPP bootstrapping info for alias: " << alias;
    g_database->remove_dpp_bootstrapping_info(alias);
}
